/// silent-payment send. The actual on-chain CreateCoin is emitted at finish
/// time by Spends::finish_with_keys's chip-0057 SP branch, which derives the
/// one-time puzzle hash from the recorded SilentPaymentPending entry. Memos
/// passed here land in the on-chain CreateCoin.memos field unchanged and are
/// visible to anyone holding the recipient's scan key.
#[cfg(feature = "chip-0057")]
fn spend_silent_payment(
    ctx: &mut SpendContext,
    spends: &mut Spends,
    recipient: &SilentPaymentAddress,
    amount: u64,
    memos: Memos,
) -> Result<(), DriverError> {
    // 1. Reserve XCH parent.
    let output = Output::new(crate::BURN_PUZZLE_HASH, amount);
    let source = spends.xch.output_source(ctx, &output)?;
    let parent = &spends.xch.items[source];
    let parent_coin_id = parent.asset.coin_id();
    let parent_puzzle_hash = parent.asset.full_puzzle_hash();

    // 2. Per-scan_pk k counter.
