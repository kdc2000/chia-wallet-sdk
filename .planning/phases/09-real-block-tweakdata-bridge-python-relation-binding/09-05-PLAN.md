        /// (post-decompression). The canonical entry point for CHIP-0058
        /// transport clients and any wallet processing real blocks (testnet11,
        /// mainnet) — not just the in-process simulator.
        ///
        /// Delegates to the driver-side

        /// which implements Pass 2a (same-puzzle-hash bucketing) and Pass 2b
        /// (iterative Tarjan SCC over opcode-64 `AssertConcurrentSpend` edges)
        /// grouping. See the driver-side module-level docs for the full
        /// grouping algorithm and the CHIP-0057 §459 identity-element guard.
        pub fn tweak_data_from_block_spends(
            coin_spends: Vec<chia_protocol::CoinSpend>,
            additions: Vec<chia_protocol::Coin>,
        ) -> Result<TweakData> {
            let driver_td = chia_sdk_driver::silent_payments::tweak_data_from_block_spends(

                &additions,

        }
